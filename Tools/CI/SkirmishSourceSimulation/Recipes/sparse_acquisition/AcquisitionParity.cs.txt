internal static class AcquisitionParity
{
    private static TargetAcquisitionInput Input(bool canAttack= true)=>new TargetAcquisitionInput{Source=new TargetIdentity(1,1),Position=new TargetPosition(1.5f,0,1.5f),GridCell=new TargetCell(1,1),GridWidth=10,GridHeight=10,AggroRangeCells=2,CellSize=1,AttackRange=2,FactionId=1,CanAttack=(byte)(canAttack?1:0),AutoEngage=1};
    private static string Run(string name,AcquisitionContext context,TargetAcquisitionInput input)
    {
        var result=TargetAcquisitionKernel.AcquireFromReadOnlyFacts(ref context,input);
        return name+"|"+(result.Acquired?1:0)+"|"+result.Target.Index+"|"+(result.Retaliation?1:0)+"|"+(result.RemoveRecentAttacker?1:0)+"|"+string.Join(",",context.Visited)+"|"+string.Join(",",context.Buildings);
    }
    private static AcquisitionContext New(bool attackMove=false,bool manual=false,bool scan=false)
    {var c=new AcquisitionContext{AttackMove=attackMove,Manual=manual,PathFollow=manual,Scan=scan};c.Initialize(new TargetIdentity(1,1));return c;}
    public static int Main()
    {
        var c=New();c.Add(20,2,1,2);c.Add(22,2,1,2);c.Add(21,0,1,2);Console.WriteLine(Run("mobile-tie",c,Input()));
        c=New(true);c.Add(30,2,1,2,building:true);c.Add(31,0,1,2,building:true);Console.WriteLine(Run("building-tie",c,Input()));
        c=New(true);c.Add(20,2,1,2);c.Add(30,2,1,2,building:true);Console.WriteLine(Run("mobile-before-building",c,Input()));
        c=New(false,true);c.Add(40,2,1,2);c.RecentId=40;Console.WriteLine(Run("manual-cancel",c,Input()));
        c=New(false,true,true);c.Add(50,2,1,2);Console.WriteLine(Run("scan-command",c,Input()));
        c=New(true,true);c.Add(60,2,1,2);Console.WriteLine(Run("attack-move-command",c,Input()));
        c=New();c.Add(70,2,1,2);c.RecentId=70;Console.WriteLine(Run("recent-retaliation",c,Input()));
        return 0;
    }
}

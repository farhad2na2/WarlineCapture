using System;
using System.Reflection;
using System.IO;
using System.Collections.Generic;
using UnityEngine.EventSystems;
using System.Threading.Tasks;
using Game.Editor;
using Game.Composition;
using Game.UI.Runtime;
using Game.Tactical.Contracts;

using UnityEngine;

public static class CampaignSelectionJourney1042
{
    static AriaTouchInputUiSystemHelper touch;
    static TaskCompletionSource<int> completion;
    static int frame=-1, gestures;
    static double due; static int cameraStep; static Vector3 cameraDestination; static double cameraSince; static int cameraRecoveryCount;
    static bool failed;static int activeMission,cardGestures,cardCaptureEntry;
    static string originalSelectionDiagnostics;
    static double staleSince; static int staleLesson=-1;
    static double maxStale;static int staleDeferrals;
    static readonly Type probe=typeof(CampaignTutorialEntryEditorProbe);
    static object Read(string name)=>probe.GetField(name,BindingFlags.Static|BindingFlags.NonPublic).GetValue(null);
    public static string Run()=>"Loaded validation-only native selection companion; no journey executed";
    public static Task<int> RunM01()=>Start(1);
    public static Task<int> RunM02()=>Start(2);
    public static Task<int> RunM03()=>Start(3);
    public static Task<int> RunM04()=>Start(4);
    public static Task<int> RunM05()=>Start(5);
    static Task<int> Start(int mission)
    {
        if(completion!=null&&!completion.Task.IsCompleted) throw new InvalidOperationException("Already active");
        failed=false;activeMission=mission;cardGestures=0;cardCaptureEntry=-1; gestures=0; frame=-1; due=0; touch=null;cameraStep=0;staleSince=0;staleLesson=-1;maxStale=0;staleDeferrals=0;cameraSince=0;cameraRecoveryCount=0;
        foreach(var name in new[]{"deadline","waitUntil","lastSkip","missingDefenseInstructionSince","missingDefenseStatusSince"})
            probe.GetField(name,BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,0d);
        foreach(var name in new[]{"lastStableStep","trainingWaitEntry","gatePoseCheckedEntry","defenseStatusCapture"})
            probe.GetField(name,BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,-1);
        completion=new TaskCompletionSource<int>();
        foreach(var assembly in AppDomain.CurrentDomain.GetAssemblies()){var observer=assembly.GetType("CampaignSelectionReadOnly1037");if(observer!=null){observer.GetMethod("Install").Invoke(null,new object[]{"/private/tmp/delegation-t7-campaign-m"+mission.ToString("00")+"-1042-observation.jsonl"});break;}}
        Environment.SetEnvironmentVariable("WARLINE_ARIA_BACKGROUND_VALIDATION","1");
        originalSelectionDiagnostics=Environment.GetEnvironmentVariable("WARLINE_SELECTION_CLICK_DIAGNOSTICS");
        Environment.SetEnvironmentVariable("WARLINE_SELECTION_CLICK_DIAGNOSTICS","1");
        probe.GetMethod("RunM"+mission.ToString("00")+"Journey").Invoke(null,null);
        // The companion owns native contact completion before the generic replay acts.
        // Reorder only the two owned QA callbacks; no gameplay callbacks are changed.
        var generic=(Canvas.WillRenderCanvases)Delegate.CreateDelegate(typeof(Canvas.WillRenderCanvases),
            probe.GetMethod("TickGameView",BindingFlags.Static|BindingFlags.NonPublic));
        Canvas.willRenderCanvases-=Frame;
        Canvas.willRenderCanvases-=generic;
        Canvas.willRenderCanvases+=Frame;
        Canvas.willRenderCanvases+=generic;
        EditorApplication.update+=Finish;
        return completion.Task;
    }
    static void Frame()
    {
        if(failed||!EditorApplication.isPlaying||(int)Read("stage")!=2)return;
        var aria=UnityEngine.Object.FindAnyObjectByType<AriaTutorialBriefingView>(FindObjectsInactive.Include);
        var canvas=aria?.GetComponentInParent<Canvas>()?.rootCanvas;
        if(canvas==null||Mathf.Abs(canvas.pixelRect.width-Screen.width)>1)return;
        bool firstFrame=Time.frameCount!=frame;
        if(firstFrame)frame=Time.frameCount;
        try
        {
            if(firstFrame)touch?.Tick(Time.unscaledTime);
            if(touch!=null&&(touch.PlayerInterrupted||touch.UnexpectedSamples!=0||touch.PhysicalInterventions!=0))
                throw new InvalidOperationException("Selection native Touch audit failed");
            
            if(cameraStep!=0)
            {
                if(EditorApplication.timeSinceStartup-cameraSince>15)throw new InvalidOperationException("Normal-input FullMap camera recovery exceeded 15 seconds");
                probe.GetField("waitUntil",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,EditorApplication.timeSinceStartup+.15);
                if(touch?.IsBusy==true||EditorApplication.timeSinceStartup<due)return;
                if(firstFrame)CameraOnly(cameraDestination);
                return;
            }
            bool hasTarget=UiShellRuntimeGateway.TryReadMissionTutorialTarget(out var target);
            if(!hasTarget)return;
            if(!UiShellRuntimeGateway.TryReadMatchHudCommandState(out var commands))return;
            var match=UnityEngine.Object.FindAnyObjectByType<MatchSceneView>();
            object bootstrap=match?.GetType().GetProperty("MatchBootstrap",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(match);
            object menu=bootstrap?.GetType().GetProperty("MainMenu").GetValue(bootstrap);
            if(menu==null)return;
            object assistant=menu.GetType().GetField("_matchHudAssistantUiSystem",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(menu);
            object highlight=assistant?.GetType().GetField("_highlightPresentationSystem",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(assistant);
            if(highlight==null)return;
            bool stale=ObserveStaleSelectionCue(highlight,target,commands);
            if(touch?.IsBusy==true)
            {
                // A second driver must not toggle controls during a native contact.
                probe.GetField("waitUntil",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,EditorApplication.timeSinceStartup+.1);
                return;
            }
            if(stale)return;
            if(EditorApplication.timeSinceStartup<due)
            {
                probe.GetField("waitUntil",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,due);
                return;
            }
            if(!firstFrame)return;
            if(!target.NeedsSelection)
            {
                if(commands.ActiveCommandMode is TacticalCommandMode.Move or TacticalCommandMode.Attack or TacticalCommandMode.Board)
                {
                    Vector3 point=match.WorldCamera.WorldToScreenPoint(target.Destination);
                    var hits=new List<RaycastResult>(); EventSystem.current?.RaycastAll(new PointerEventData(EventSystem.current){position=point},hits);
                    var popup=UnityEngine.Object.FindAnyObjectByType<MatchHudFullMapPopupView>(FindObjectsInactive.Include);
                    if(cameraStep!=0||popup!=null&&popup.IsOpen||point.z<=0||!Screen.safeArea.Contains(point)||hits.Count>0)CameraOnly(target.Destination);
                }
                return;
            }
            if(activeMission==1&&target.DragSelection)
            {
                if(commands.ActiveCommandMode!=TacticalCommandMode.Select)return;
                var tray=UnityEngine.Object.FindAnyObjectByType<MatchHudSquadTrayView>();
                var button=tray?.VisibleCardButton(0);
                var guideRect=(RectTransform)highlight.GetType().GetField("_directTutorialTarget",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(highlight);
                if(guideRect!=button?.transform)return;
                int currentEntry=(int)Read("entry");
                if(cardCaptureEntry!=currentEntry)
                {
                    cardCaptureEntry=currentEntry;probe.GetMethod("CaptureJourney",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,new object[]{"presented-rifle-card-fallback"});
                    due=EditorApplication.timeSinceStartup+.6;probe.GetField("waitUntil",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,due);
                    Debug.Log("[CampaignNativeSelection] actualCardGuideObserved=1 body="+aria.BodyText.text+" inputInjected=0");return;
                }
                var qaArgs=new object[]{button,Vector2.zero};
                if(button!=null&&button.interactable&&(bool)typeof(SkirmishS002AriaRunHarness).GetMethod("TryPresentedButtonPoint",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,qaArgs))
                {
                    var point=(Vector2)qaArgs[1];if(Gesture(point,point,0)){cardGestures++;Debug.Log("[CampaignNativeSelection] input=Touch presentedRifleSquadCard=1 required="+target.RequiredSelectionCount+" cardGestures="+cardGestures+" actorOrSelectionInjected=0 actualPresentedCardFallback=Verified");}return;
                }
                throw new InvalidOperationException("Normal presented rifle squad card unavailable");
            }
            if(commands.ActiveCommandMode!=TacticalCommandMode.Select)return;
            Vector2 start,end;
            var args=new object[]{Vector2.zero,Vector2.zero};
            bool visibleDrag=(bool)highlight.GetType().GetMethod("TryObserveVisibleSelectionDrag",BindingFlags.Instance|BindingFlags.NonPublic).Invoke(highlight,args);
            if(visibleDrag)
            {
                start=(Vector2)args[0];end=(Vector2)args[1];
            }
            else
            {
                if(target.DragSelection)
                {
                    if(++cameraRecoveryCount>4)throw new InvalidOperationException("Group selection guide remains unavailable after four normal FullMap camera recoveries");
                    Debug.Log("[CampaignNativeSelection] visibleGroupDragUnavailable=1 publicRequired="+target.RequiredSelectionCount+" publicSelection="+target.Selection+" tapFallback=0");
                    cameraDestination=target.Selection;cameraSince=EditorApplication.timeSinceStartup;
                    CameraOnly(cameraDestination);
                    return;
                }
                if(!hasTarget||!target.NeedsSelection)return;
                var world=match.WorldCamera.WorldToScreenPoint(target.Selection);
                if(world.z<=0)return;
                start=end=world;
                if(!Screen.safeArea.Contains(start))return;
                var hits=new List<RaycastResult>();
                EventSystem.current?.RaycastAll(new PointerEventData(EventSystem.current){position=start},hits);
                if(hits.Count>0)return; // A world selection tap must never activate a HUD command.

            }
            if(touch==null){touch=new AriaTouchInputUiSystemHelper();if(!touch.Start())throw new InvalidOperationException("Native Touch unavailable");}
            if(!touch.TryGesture(start,end,visibleDrag?1.2f:.3f,visibleDrag?.8f:0f,Time.unscaledTime))return;
            gestures++;cameraRecoveryCount=0;due=EditorApplication.timeSinceStartup+3;
            probe.GetField("waitUntil",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,due);
            Debug.Log("[CampaignNativeSelection] input=Touch drag="+target.DragSelection+" expected="+target.RequiredSelectionCount+" gesture="+gestures+" start="+start+" end="+end+" actorOrSelectionInjected=0");
        }
        catch(Exception ex){failed=true;probe.GetField("failure",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,"Native selection companion: "+ex);Debug.LogException(ex);}
    }
    static bool ObserveStaleSelectionCue(object highlight, Game.UI.Contracts.UiMissionTutorialTarget target,
        Game.UI.Contracts.UiMatchHudCommandStateModel commands)
    {
        var rect=(RectTransform)highlight.GetType().GetField("_directTutorialTarget",BindingFlags.Instance|BindingFlags.NonPublic).GetValue(highlight);
        var button=rect?.GetComponent<UnityEngine.UI.Button>();
        bool stale=GameObject.Find("AriaAssistantTargetIndicatorRuntime")!=null && button?.name=="SelectCommand" && !target.NeedsSelection;
        double now=EditorApplication.timeSinceStartup;
        if(!stale)
        {
            if(staleSince!=0){maxStale=Math.Max(maxStale,now-staleSince);Debug.Log("[CampaignNativeSelection] stale-cue-settled seconds="+(now-staleSince).ToString("F3"));}
            staleSince=0;staleLesson=-1;return false;
        }
        int lesson=(int)Read("journeyStep");
        if(staleSince==0||staleLesson!=lesson)
        {
            staleSince=now;staleLesson=lesson;staleDeferrals++;
            probe.GetMethod("CaptureJourney",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,new object[]{"lesson-"+lesson+"-stale-selection-cue"});
            Debug.Log("[CampaignNativeSelection] stale-cue-deferred lesson="+lesson+" needsSelection=0 mode="+commands.ActiveCommandMode+" actionInjected=0");
        }
        if(now-staleSince>3)throw new InvalidOperationException("Persistent Select cue while public selection is satisfied for over 3 seconds; lesson="+lesson);
        probe.GetField("waitUntil",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,now+.1);
        return true;
    }
    static bool Gesture(Vector2 from,Vector2 to,float travel)
    {
        if(touch==null){touch=new AriaTouchInputUiSystemHelper();if(!touch.Start())throw new InvalidOperationException("Native camera Touch unavailable");}
        if(!touch.TryGesture(from,to,.15f,travel,Time.unscaledTime))return false;
        gestures++;due=EditorApplication.timeSinceStartup+1.25;
        probe.GetField("waitUntil",BindingFlags.Static|BindingFlags.NonPublic).SetValue(null,due);
        return true;
    }
    static void CameraOnly(Vector3 destination)
    {
        var popup=UnityEngine.Object.FindAnyObjectByType<MatchHudFullMapPopupView>(FindObjectsInactive.Include);
        if(popup==null||!popup.IsOpen)
        {
            cameraStep=0;
            foreach(var map in UnityEngine.Object.FindObjectsByType<MatchHudMinimapView>(FindObjectsSortMode.None))
            {
                if(map.GetComponentInParent<MatchHudFullMapPopupView>()!=null||map.MapRect==null)continue;
                var canvas=map.GetComponentInParent<Canvas>();Camera cam=canvas!=null&&canvas.renderMode!=RenderMode.ScreenSpaceOverlay?canvas.worldCamera:null;
                Vector2 point=RectTransformUtility.WorldToScreenPoint(cam,map.MapRect.TransformPoint(map.MapRect.rect.center));
                if(Gesture(point,point,0)){cameraStep=1;cameraDestination=destination;cameraSince=EditorApplication.timeSinceStartup;Debug.Log("[CampaignNativeCamera] phase=Open input=Touch troopOrderInjected=0");}
                return;
            }
            return;
        }
        if(cameraStep==2)
        {
            var args=new object[]{popup.CloseAction,Vector2.zero};
            if(!(bool)typeof(SkirmishS002AriaRunHarness).GetMethod("TryPresentedButtonPoint",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,args))return;
            if(Gesture((Vector2)args[1],(Vector2)args[1],0)){cameraStep=0;Debug.Log("[CampaignNativeCamera] phase=Close input=Touch troopOrderInjected=0");}
            return;
        }
        var full=popup.Minimap;if(full==null)return;
        var root=full.GetComponentInParent<Canvas>();Camera camera=root!=null&&root.renderMode!=RenderMode.ScreenSpaceOverlay?root.worldCamera:null;
        if(!full.TryGetPresentedMapPoint(destination,camera,out var goal))return;
        bool drag=full.ViewportRect!=null&&RectTransformUtility.RectangleContainsScreenPoint(full.ViewportRect,goal,camera);
        Vector2 start=drag?RectTransformUtility.WorldToScreenPoint(camera,full.ViewportRect.TransformPoint(full.ViewportRect.rect.center)):goal;
        Vector2 end=drag&&(start-goal).sqrMagnitude<=25?goal+Vector2.up*12:goal;
        if(Gesture(start,end,drag?.45f:0)){cameraStep=2;Debug.Log("[CampaignNativeCamera] phase=Focus input=FullMapTouch publicDestination="+destination+" troopOrderInjected=0");}
    }
    static void Finish()
    {
        if(!(bool)Read("finished"))return;
        Canvas.willRenderCanvases-=Frame;EditorApplication.update-=Finish;
        bool pass=!failed&&Read("failure")==null&&gestures>0&&touch!=null&&touch.CompletedGestures==gestures&&touch.UnexpectedSamples==0&&touch.PhysicalInterventions==0;
        Debug.Log("[CampaignNativeSelection] result="+(pass?"Passed":"Failed")+" gestures="+gestures+" completed="+(touch?.CompletedGestures??0)+" unexpected="+(touch?.UnexpectedSamples??0)+" interventions="+(touch?.PhysicalInterventions??0)+" staleCueDeferrals="+staleDeferrals+" maximumStaleSeconds="+maxStale.ToString("F3")+" staleCueBoundSeconds=3 fullJourneyStrictTouchAudit=0");
        touch?.Dispose();touch=null;
        Environment.SetEnvironmentVariable("WARLINE_SELECTION_CLICK_DIAGNOSTICS",originalSelectionDiagnostics);
        completion.TrySetResult(pass?0:1);
    }
}

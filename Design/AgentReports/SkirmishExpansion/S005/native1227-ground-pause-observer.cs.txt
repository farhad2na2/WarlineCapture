    static bool Case(string name,bool engaged,bool vehicle,bool near,bool inRange,bool active,float delta,bool positive,bool conservation)
    {
        using var f=new Fixture(engaged,vehicle,near,inRange);var em=f.Em;
        var before=em.GetComponentData<LocalTransform>(f.Actor);var grid=em.GetComponentData<UnitGrid>(f.Actor);var motion=em.GetComponentData<UnitVehicleKinematics>(f.Actor);
        bool path=em.HasComponent<UnitPathFollow>(f.Actor),target=em.HasComponent<UnitTarget>(f.Actor);int index=path?em.GetComponentData<UnitPathFollow>(f.Actor).PathIndex:-1;
        f.Step(active,delta);
        var after=em.GetComponentData<LocalTransform>(f.Actor);bool hasPath=em.HasComponent<UnitPathFollow>(f.Actor);int nowIndex=hasPath?em.GetComponentData<UnitPathFollow>(f.Actor).PathIndex:-1;
        var row=new Row{scenario=name,system=engaged?"UnitEngagedMovementSystem":"UnitGridMovementSystem",mode=conservation?"Conservation":"BeforeFix",delta=delta,simulationActive=active,positionDelta=math.distance(before.Position,after.Position),rotationDelta=math.distance(before.Rotation.value,after.Rotation.value),transformChanged=!before.Equals(after),gridChanged=!grid.Equals(em.GetComponentData<UnitGrid>(f.Actor)),kinematicsChanged=!motion.Equals(em.GetComponentData<UnitVehicleKinematics>(f.Actor)),pathChanged=path!=hasPath||index!=nowIndex,targetChanged=target!=em.HasComponent<UnitTarget>(f.Actor),originalPathIndex=index,currentPathIndex=nowIndex,positiveControl=positive};
        Debug.Log("[GroundPauseSystem1227Case] "+JsonUtility.ToJson(row));
        if(positive){if(!row.transformChanged||(inRange?(row.positionDelta!=0||row.rotationDelta<=.01f):row.positionDelta<=.01f))throw new InvalidOperationException("Positive native control did not perform intended facing/movement:"+name);return false;}

        else if(engaged){if(row.positionDelta!=0||row.rotationDelta<=.01f)throw new InvalidOperationException("Expected native zero-delta engaged-facing mutation not reproduced:"+name);}

        return row.transformChanged;
    }

    {
        if(string.IsNullOrEmpty(source))return "";

        if(parts.Length!=2 || !int.TryParse(parts[0],out int minutes) || !int.TryParse(parts[1],out int seconds) || minutes<0 || seconds<0 || seconds>=60)return "";
        return value;

        Write(new Row { stage="ProbeClosed" });
    }
    static void Update()
    {
        try
        {
            if (UnityEditor.EditorApplication.isPlaying) everPlaying=true;
            if (everPlaying && !UnityEditor.EditorApplication.isPlayingOrWillChangePlaymode) { End(); return; }
            if (!UnityEditor.EditorApplication.isPlaying || UnityEditor.EditorApplication.timeSinceStartup<nextSample) return;
            nextSample=UnityEditor.EditorApplication.timeSinceStartup+.10d;
            Snapshot("FrameSample", default);
        }
        catch (System.Exception e) { Write(new Row { stage="ProbeFailure", error=e.ToString() }); End(); }
    }
    static void Touch(Game.UI.Runtime.AriaTouchInputUiSystemHelper actuator, string phase, UnityEngine.Vector2 point)
    {
        try
        {
            if (!UnityEditor.EditorApplication.isPlaying) return;
            if (phase=="Began") gestureSequence++;
            if (phase=="Dispatched" || phase=="Began" || phase=="Moved" || phase=="Ended" || phase=="Canceled") Snapshot("Touch"+phase,point);
        }
        catch (System.Exception e) { Write(new Row { stage="ProbeFailure", error=e.ToString() }); End(); }
    }

        if (scroll==null || scroll.content==null || scroll.viewport==null || catalog==null) return;
        var world=Unity.Entities.World.DefaultGameObjectInjectionWorld;
        if (world==null || !world.IsCreated) return;

        using (var q=em.CreateEntityQuery(typeof(Game.UI.Shell.Contracts.Ecs.AriaSkirmishObservationComponent)))
        {
            if (q.CalculateEntityCount()!=1) return;
            observation=q.GetSingleton<Game.UI.Shell.Contracts.Ecs.AriaSkirmishObservationComponent>().Value;
        }

        using (var q=em.CreateEntityQuery(typeof(Game.UI.Shell.Contracts.Ecs.AriaPlaySessionComponent)))
        {
            if (q.CalculateEntityCount()!=1) return;
            play=q.GetSingleton<Game.UI.Shell.Contracts.Ecs.AriaPlaySessionComponent>();
        }

            play.TargetId==observation.RecruitAntiAir.Id ? catalog.ResolveCatalogTarget(Game.UI.Contracts.BuildDrawerCategory.Vehicles,"Unit_Veh_Missle_Launcher_Air") : null;

            dragEndX=observation.RecruitTanker.DragEnd.x, dragEndY=observation.RecruitTanker.DragEnd.y,

            viewportWidth=scroll.viewport.rect.width, viewportHeight=scroll.viewport.rect.height, scrollVelocityY=scroll.velocity.y };
        var contentBounds=BoundsOf(scroll.content,scroll.viewport); var viewportBounds=scroll.viewport.rect;
        row.contentBoundsMinX=contentBounds.xMin;row.contentBoundsMinY=contentBounds.yMin;row.contentBoundsMaxX=contentBounds.xMax;row.contentBoundsMaxY=contentBounds.yMax;
        row.viewportBoundsMinX=viewportBounds.xMin;row.viewportBoundsMinY=viewportBounds.yMin;row.viewportBoundsMaxX=viewportBounds.xMax;row.viewportBoundsMaxY=viewportBounds.yMax;

            var corners=new UnityEngine.Vector3[4];targetRect.GetWorldCorners(corners);
            var low=UnityEngine.RectTransformUtility.WorldToScreenPoint(camera,corners[0]);var high=UnityEngine.RectTransformUtility.WorldToScreenPoint(camera,corners[2]);
            row.targetScreenMinX=UnityEngine.Mathf.Min(low.x,high.x);row.targetScreenMinY=UnityEngine.Mathf.Min(low.y,high.y);

                row.firstRaycastScrollOwner=PathOf(hitScroll?.transform);
                row.firstRaycastMatchesCatalogScroll=hitScroll==scroll;
                row.firstRaycastCamera=first.module?.eventCamera?.name??"";
            }
        }
        Write(row);
    }
    static UnityEngine.Rect BoundsOf(UnityEngine.RectTransform target,UnityEngine.RectTransform reference)
    {
        var corners=new UnityEngine.Vector3[4];target.GetWorldCorners(corners);
        var min=new UnityEngine.Vector2(float.MaxValue,float.MaxValue);var max=new UnityEngine.Vector2(float.MinValue,float.MinValue);
        foreach(var corner in corners) {var p=(UnityEngine.Vector2)reference.InverseTransformPoint(corner);min=UnityEngine.Vector2.Min(min,p);max=UnityEngine.Vector2.Max(max,p);}
        return UnityEngine.Rect.MinMaxRect(min.x,min.y,max.x,max.y);
    }
    static string PathOf(UnityEngine.Transform t)=>t==null?"":t.parent==null?t.name:PathOf(t.parent)+"/"+t.name;
    static void Write(Row row)
    {
        if (rows++>6000) return;
        row.frame=UnityEngine.Time.frameCount;row.time=UnityEngine.Time.time;
        System.IO.File.AppendAllText(Evidence,UnityEngine.JsonUtility.ToJson(row)+"\n");
    }

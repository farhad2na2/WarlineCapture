    static string ClockNumber(string source)
    {
        if(string.IsNullOrEmpty(source))return "";
        int separator=source.LastIndexOf("  ",StringComparison.Ordinal);
        string value=separator>=0 ? source.Substring(separator+2).Trim() : source.Trim();
        var parts=value.Split(':');
        if(parts.Length!=2 || !int.TryParse(parts[0],out int minutes) || !int.TryParse(parts[1],out int seconds) || minutes<0 || seconds<0 || seconds>=60)return "";
        return value;
    }

    static void PerfStart()
    {
        if(File.Exists(PerfEvidence))throw new IOException("Preserve existing1183 performance evidence");
        using(File.Open(PerfEvidence,FileMode.CreateNew,FileAccess.Write,FileShare.Read)){}
        perfFrames=new PerfFrame[PerfCapacity];perfMarkerValues=new long[PerfCapacity,PerfMarkerBase.Length];perfCensuses=new PerfCensus[1805];
        perfMarkers=new Unity.Profiling.ProfilerRecorder[PerfMarkerBase.Length];perfActualNames=new string[PerfMarkerBase.Length];perfMarkerErrors=new string[PerfMarkerBase.Length];perfMarkerAttempts=new int[PerfMarkerBase.Length];
        perfCount=perfCensusCount=perfActiveWarmup=perfEditorDistinct=0;perfLastFrame=perfLastEditorFrame=-1;
        perfDone=perfOverflow=false;perfFailure=perfGcError=perfAttempt=perfWorldName="";perfStarted=EditorApplication.timeSinceStartup;perfNextCensus=0;perfPreviousWall=0;
        perfPhase=0;perfElapsed=0;perfPlayingMatch=false;perfPeakMemory=perfPeakMono=0;
        try{perfPreviousThread=GC.GetAllocatedBytesForCurrentThread();perfThreadAvailable=true;}catch(Exception e){perfThreadAvailable=false;perfFailure="ThreadCounterUnavailable:"+e.GetType().Name;}

    static void PerfLifecycle()
    {
        if(perfDone || perfFrames==null)return;
        if(EditorApplication.isPlaying && Time.frameCount!=perfLastEditorFrame){perfLastEditorFrame=Time.frameCount;perfEditorDistinct++;}
    }
    static void PerfRecorderRetry(string world)
    {
        for(int i=0;i<PerfMarkerBase.Length;i++)
        {
            if(perfMarkers[i].Valid)continue;
            string name=i>=11 ? world+" "+PerfMarkerBase[i] : PerfMarkerBase[i];
            perfActualNames[i]=name;perfMarkerAttempts[i]++;
            try{perfMarkers[i].Dispose();perfMarkers[i]=Unity.Profiling.ProfilerRecorder.StartNew(Unity.Profiling.ProfilerCategory.Scripts,name);if(!perfMarkers[i].Valid)perfMarkerErrors[i]="NativeRecorderInvalid";else perfMarkerErrors[i]="";}
            catch(Exception e){perfMarkerErrors[i]=e.GetType().Name+":"+e.Message;}
        }
        if(!perfGc.Valid)
        {
            try{perfGc.Dispose();perfGc=Unity.Profiling.ProfilerRecorder.StartNew(Unity.Profiling.ProfilerCategory.Memory,"GC Allocated In Frame");perfGcError=perfGc.Valid?"":"NativeRecorderInvalid";}
            catch(Exception e){perfGcError=e.GetType().Name+":"+e.Message;}
        }
    }
    static void PerfReadCensus(double wall)
    {
        if(perfCensusCount>=perfCensuses.Length){perfOverflow=true;perfFailure="CensusCapacityExceeded";return;}
        var c=new PerfCensus{frame=Time.frameCount,editorWall=wall,focused=Application.isFocused,screenWidth=Screen.width,screenHeight=Screen.height,qualityLevel=QualitySettings.GetQualityLevel(),antiAliasing=QualitySettings.antiAliasing,vSync=QualitySettings.vSyncCount,targetFrameRate=Application.targetFrameRate,simulationTimeScale=Time.timeScale};
        var names=QualitySettings.names;c.qualityName=c.qualityLevel>=0&&c.qualityLevel<names.Length?names[c.qualityLevel]:"Unknown";
        var pipeline=UnityEngine.Rendering.GraphicsSettings.currentRenderPipeline;c.pipeline=pipeline==null?"BuiltIn":pipeline.name;
        var safe=Screen.safeArea;c.safeX=safe.x;c.safeY=safe.y;c.safeWidth=safe.width;c.safeHeight=safe.height;
        perfPeakMemory=Math.Max(perfPeakMemory,UnityEngine.Profiling.Profiler.GetTotalAllocatedMemoryLong()/1048576d);perfPeakMono=Math.Max(perfPeakMono,UnityEngine.Profiling.Profiler.GetMonoUsedSizeLong()/1048576d);c.peakAllocatedMB=perfPeakMemory;c.peakMonoMB=perfPeakMono;
        var w=World.DefaultGameObjectInjectionWorld;perfPlayingMatch=false;perfPhase=bindingSeen?6:0;perfElapsed=0;perfUnits=perfStructures=perfProjectiles=perfPaths=perfEngaged=0;
        if(w!=null&&w.IsCreated)
        {
            c.world=w.Name;perfWorldName=w.Name;PerfRecorderRetry(w.Name);var em=w.EntityManager;
            using(var q=em.CreateEntityQuery(typeof(SkirmishExpandedSessionComponent)))
            {
                if(q.CalculateEntityCount()>1)throw new InvalidOperationException("Performance observed multiple expanded sessions");
                if(q.CalculateEntityCount()==1)
                {
                    var e=q.GetSingletonEntity();var s=em.GetComponentData<SkirmishExpandedSessionComponent>(e);
                    if(!s.CatalogId.Equals(new Unity.Collections.FixedString64Bytes("S005")))throw new InvalidOperationException("Performance observed foreign catalog");
                    string identity=s.SessionId.ToString();if(perfAttempt.Length>0&&perfAttempt!=identity)throw new InvalidOperationException("Performance attempt identity changed during one normal journey");perfAttempt=identity;c.attempt=identity;c.hasActualSession=true;c.nativePhase=s.Phase.ToString();
                    if(em.HasComponent<SkirmishMatchState>(e)){var m=em.GetComponentData<SkirmishMatchState>(e);c.hasActualMatch=true;perfElapsed=m.ElapsedSeconds;perfPlayingMatch=m.Phase==SkirmishPhase.Playing;perfPhase=m.Phase==SkirmishPhase.Finished?5:1;}
                    if(em.HasBuffer<SkirmishContactLedgerElement>(e))c.ledgerEntries=em.GetBuffer<SkirmishContactLedgerElement>(e,true).Length;
                    using(var actors=em.CreateEntityQuery(typeof(SkirmishAttemptOwnedComponent),typeof(UnitHealth)))
                    using(var entities=actors.ToEntityArray(Unity.Collections.Allocator.Temp))
                    {
                        for(int i=0;i<entities.Length;i++)
                        {
                            var actor=entities[i];var owner=em.GetComponentData<SkirmishAttemptOwnedComponent>(actor);if(!owner.SessionId.Equals(s.SessionId))continue;c.healthOwnedEntities++;
                            if(em.GetComponentData<UnitHealth>(actor).Current<=0)continue;
                            if(owner.IsStructure!=0)c.structures++;else c.units++;
                            if(em.HasComponent<SelectedUnitTag>(actor))c.selectedEntities++;
                            if(em.HasComponent<UnitPathFollow>(actor))c.pathFollowing++;
                            if(em.HasComponent<EngageTarget>(actor)&&em.GetComponentData<EngageTarget>(actor).Target!=Entity.Null)c.engagements++;
                        }
                    }
                    // The optional knowledge buffer is on actual contacts (FogService.EnsureKnowledge), not the session.
                    // Count every current-attempt contact with that buffer, independently of live-health census membership.
                    using(var contacts=em.CreateEntityQuery(typeof(SkirmishAttemptOwnedComponent),typeof(SkirmishContactKnowledgeElement)))
                    using(var entities=contacts.ToEntityArray(Unity.Collections.Allocator.Temp))
                    {
                        for(int i=0;i<entities.Length;i++)
                        {
                            var contact=entities[i];if(!em.GetComponentData<SkirmishAttemptOwnedComponent>(contact).SessionId.Equals(s.SessionId))continue;
                            c.knowledgeContactEntities++;c.knowledgeEntries+=em.GetBuffer<SkirmishContactKnowledgeElement>(contact,true).Length;
                        }
                    }
                    // Scene aggregate projectile count is expressly not attempt attribution.
                    using(var air=em.CreateEntityQuery(typeof(AirMissileProjectileComponent)))c.projectiles+=air.CalculateEntityCount();
                    using(var ground=em.CreateEntityQuery(typeof(GroundMissileProjectileComponent)))c.projectiles+=ground.CalculateEntityCount();
                }
            }
        }
        c.elapsed=perfElapsed;perfUnits=c.units;perfStructures=c.structures;perfProjectiles=c.projectiles;perfPaths=c.pathFollowing;perfEngaged=c.engagements;c.phase=perfPhase;perfCensuses[perfCensusCount++]=c;
    }
    static void PerfSampleRendered()
    {
        if(perfDone || perfFrames==null || perfLastFrame==Time.frameCount)return;
        long entered=System.Diagnostics.Stopwatch.GetTimestamp();double wall=EditorApplication.timeSinceStartup;
        if(wall>=perfNextCensus){PerfReadCensus(wall);perfNextCensus=wall+1d;}
        if(perfCount>=PerfCapacity){perfOverflow=true;perfFailure="RenderedSampleCapacityExceeded";return;}
        int frame=Time.frameCount;int gap=perfLastFrame<0?0:frame-perfLastFrame-1;
        if(gap<0)throw new InvalidOperationException("Native frame counter regressed");
        if(perfPlayingMatch)perfActiveWarmup++;
        bool warm=perfPlayingMatch&&perfActiveWarmup>180&&perfElapsed>=15f;
        int phase=returnSeen?6:finishedVictorySeen?5:perfPlayingMatch?(warm?(perfEngaged>0?4:3):2):perfPhase;
        long allocated=-1;if(perfThreadAvailable){long now=GC.GetAllocatedBytesForCurrentThread();allocated=Math.Max(0,now-perfPreviousThread);perfPreviousThread=now;}
        bool gcValid=perfGc.Valid,gcNonempty=gcValid&&perfGc.Count>0;
        var sample=new PerfFrame{frame=frame,phase=phase,frameGap=gap,editorWall=wall,wallIntervalMs=perfPreviousWall>0?(wall-perfPreviousWall)*1000d:0,frameIntervalMs=Time.unscaledDeltaTime*1000d,simulationElapsed=perfElapsed,timeScale=Time.timeScale,focused=Application.isFocused,postWarmup=warm,activeWarmupFrames=perfActiveWarmup,censusIndex=perfCensusCount-1,threadCounterAvailable=perfThreadAvailable,threadAllocatedBytes=allocated,unityGcValid=gcValid,unityGcNonempty=gcNonempty,unityAllocatedBytes=gcNonempty?perfGc.LastValue:-1};
        for(int i=0;i<perfMarkers.Length;i++)perfMarkerValues[perfCount,i]=perfMarkers[i].Valid&&perfMarkers[i].Count>0?perfMarkers[i].LastValue:-1;
        sample.callbackMs=(System.Diagnostics.Stopwatch.GetTimestamp()-entered)*1000d/System.Diagnostics.Stopwatch.Frequency;
        perfFrames[perfCount++]=sample;perfLastFrame=frame;perfPreviousWall=wall;
    }
    static void PerfFinish(bool closed,string reason)
    {
        if(perfDone || perfFrames==null)return;
        perfDone=true;
        if(closed)try{PerfReadCensus(EditorApplication.timeSinceStartup);}catch(Exception e){perfFailure="FinalCensusFailure:"+e.ToString();}
        try
        {
            using(var writer=new StreamWriter(PerfEvidence,true))
            {
                writer.WriteLine(JsonUtility.ToJson(new PerfHeader{markerRequestedNames=PerfMarkerBase,markerActualNames=perfActualNames,markerErrors=perfMarkerErrors,markerStartAttempts=perfMarkerAttempts,gcError=perfGcError}));
                for(int i=0;i<perfCensusCount;i++)writer.WriteLine(JsonUtility.ToJson(perfCensuses[i]));
                var raw=new PerfRaw{markerNanoseconds=new long[PerfMarkerBase.Length]};
                for(int f=0;f<perfCount;f++){raw.sample=perfFrames[f];for(int m=0;m<PerfMarkerBase.Length;m++)raw.markerNanoseconds[m]=perfMarkerValues[f,m];writer.WriteLine(JsonUtility.ToJson(raw));}
                writer.WriteLine(JsonUtility.ToJson(new PerfClosure{actualPlayClosed=closed,reason=reason,failure=perfFailure,overflow=perfOverflow,samples=perfCount,censuses=perfCensusCount,editorDistinctFrames=perfEditorDistinct,lastObservedEditorFrame=perfLastEditorFrame,setupSeen=setupSeen,bindingSeen=bindingSeen,resultSeen=resultSeen,returnSeen=returnSeen,threadCounterAvailable=perfThreadAvailable,startedWall=perfStarted,closedWall=EditorApplication.timeSinceStartup}));
            }
            Debug.Log("[S005Performance1183] observerClosed=1 scope=InstrumentedEditor diagnosticOnly=1 deviceCertified=0 gameplayWrites=0");
        }
        finally
        {
            for(int i=0;i<perfMarkers.Length;i++)perfMarkers[i].Dispose();perfGc.Dispose();
        }
    }

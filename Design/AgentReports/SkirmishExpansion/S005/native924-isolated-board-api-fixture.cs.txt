        var context=getContext.Invoke(closure,null);
        var pointer=closure.GetType().GetField("rtsSelectionPointerTargetCommandSystem",fields).GetValue(closure);
        var point=(Vector2)Camera.main.WorldToScreenPoint((Vector3)em.GetComponentData<LocalTransform>(carrier).Position+Vector3.up);
        object[] pick={context,point,em,Entity.Null};

        Debug.Log("[S005BoardApiPointer] actualCarrier="+carrier+" point="+point+" found="+found+" picked="+pick[3]+" scope=IsolatedProjection normalInput=0");

        object air=closure.GetType().GetField("unitTransportAirPickupSystem",fields).GetValue(closure);
        object move=closure.GetType().GetField("unitMoveOrderSystem",fields).GetValue(closure);

    }

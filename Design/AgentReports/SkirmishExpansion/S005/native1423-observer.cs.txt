    static string ClockNumber(string source)
    {
        if(string.IsNullOrEmpty(source))return "";
        int separator=source.LastIndexOf("  ",StringComparison.Ordinal);
        string value=separator>=0 ? source.Substring(separator+2).Trim() : source.Trim();
        var parts=value.Split(':');
        if(parts.Length!=2 || !int.TryParse(parts[0],out int minutes) || !int.TryParse(parts[1],out int seconds) || minutes<0 || seconds<0 || seconds>=60)return "";
        return value;
    }

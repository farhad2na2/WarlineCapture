using System;
using System.Collections.Generic;
using System.IO;
using System.Reflection;
using Game.Components;
using Game.Configs;
using Game.Editor;
using Game.Runtime;
using Game.UI.Runtime;
using TMPro;
using Unity.Entities;
using UnityEditor;
using UnityEngine;
using UnityEngine.UI;

// External QA only: existing public menu/ARIA journey is the sole actuator.
// This adapter writes evidence, never entities, orders, clocks, controls or displayed text.
public static class S005NormalAria1151Observer
{
    const string Report = "/Users/farhad/Projects/WarlineCapture/Design/AgentReports/SkirmishExpansion/S005/";
    const string Evidence = Report + "native1151-normal-aria-observation.jsonl";
    static bool everPlaying, setupSeen, bindingSeen, purposeSeen, activeControlSeen, resultSeen, returnSeen;
    static double next, began;
    static readonly HashSet<string> receiptStates = new();
    static readonly Dictionary<string,string> lastRows = new();
    static readonly MethodInfo ReadPoint = typeof(SkirmishS002AriaRunHarness).GetMethod("TryPresentedButtonPoint", BindingFlags.NonPublic | BindingFlags.Static);
    static readonly MethodInfo TracePath = typeof(SkirmishS002AriaRunHarness).GetMethod("LiveTracePath", BindingFlags.NonPublic | BindingFlags.Static);

    [Serializable] public sealed class Row
    {
        public string stage="", catalog="", definition="", size="", difficulty="", army="", start="", readiness="", objective="", attempt="", locale="", tracePath="";
        public string title="", body="", roster="", economy="", intel="", summary="", modelObjective="", clock="", playerBase="", enemyBase="", controlText="", controlPath="", role="", phase="", reason="";
        public int frame, seed, deadline, faction, producer, materialsPaid, expectedCost, members, deliveredMembers, readinessValue;
        public uint reservation;
        public bool sharedFog, developmentFullVision, renderedTitle, renderedBody, renderedObjective, renderedRoster, renderedEconomy, renderedIntel, renderedPlayerBase, renderedEnemyBase, renderedClock, presentedControl, ariaActive, finished, expanded, combinedArmsField, setupSeen, bindingSeen, purposeSeen, activeControlSeen, resultSeen, returnSeen, normalDriverFlags;
        public float timeScale;
    }
    public static void Run() => Debug.Log("[S005NormalAria1151] loadedOnly=1 gameplayStarted=0");
    public static void RunValidation()
    {
        if (EditorApplication.isPlayingOrWillChangePlaymode) throw new InvalidOperationException("Normal1151 requires idle main Editor.");
        if (ReadPoint == null || TracePath == null) throw new MissingMethodException("Required existing read-only harness observer bridge absent.");
        if (File.Exists(Evidence)) throw new IOException("Preserve existing1151 observation evidence.");
        using (File.Open(Evidence,FileMode.CreateNew,FileAccess.Write,FileShare.Read)) { }
        everPlaying=setupSeen=bindingSeen=purposeSeen=activeControlSeen=resultSeen=returnSeen=false;
        receiptStates.Clear(); lastRows.Clear(); next=0; began=EditorApplication.timeSinceStartup;
        EditorApplication.update += Observe;
        try { SkirmishS002AriaRunHarness.RunS005EditorAriaEn(); }
        catch { EditorApplication.update -= Observe; throw; }
        bool flags=Environment.GetEnvironmentVariable("WARLINE_S004_MENU_JOURNEY")=="1"
            && Environment.GetEnvironmentVariable("WARLINE_SKIRMISH_CATALOG")=="S005"
            && Environment.GetEnvironmentVariable("WARLINE_S002_LOCALE")=="en"
            && Environment.GetEnvironmentVariable("WARLINE_S002_SEED")=="104734"
            && Environment.GetEnvironmentVariable("WARLINE_TERMINAL_ACTION")=="ADJUST SETUP";
        foreach(string name in new[]{"WARLINE_SKIRMISH_STARTUP_ONLY","WARLINE_SKIRMISH_INPUT_DIAGNOSTIC","WARLINE_S005_NATIVE_RECOVERY_ONLY","WARLINE_S005_PLAYER_RESUME_PROBE","WARLINE_S005_PAID_BUILDING_CLEARANCE_ONLY","WARLINE_S004_MANUAL_PROBE","WARLINE_S004_STOP_PROBE"}) flags &= string.IsNullOrEmpty(Environment.GetEnvironmentVariable(name));
        Append(new Row{stage="NormalDriverFlags",normalDriverFlags=flags,reason="Existing MenuProbe normal path; no diagnostic/manual/resume/forced outcome path"});
        Append(new Row {stage="StartedExistingPublicJourney",reason="MenuProbe.RunS005EditorAriaEn; observerReadOnly=1; noGameplayWrites=1"});
    }
    static void Append(Row row)
    {
        string key=row.stage+"|"+row.controlPath+"|"+row.reservation+"|"+(row.stage=="NativeRenderedResultText" ? row.title : "");
        string state=JsonUtility.ToJson(row);
        if(lastRows.TryGetValue(key,out var prior) && prior==state)return;
        lastRows[key]=state;
        row.frame=Time.frameCount;row.timeScale=Time.timeScale;
        row.locale=UiShellRuntimeGateway.Localization.CurrentLocaleCode;
        File.AppendAllText(Evidence,JsonUtility.ToJson(row)+"\n");
    }
    static string PathOf(Transform t) => t==null ? "" : t.parent==null ? t.name : PathOf(t.parent)+"/"+t.name;
    static bool Presented(Button b)
    {
        if (b==null) return false;
        return (bool)ReadPoint.Invoke(null,new object[]{b,default(Vector2)});
    }
    static string Text(TMP_Text t) => t==null ? "" : t.text ?? "";
    static bool Rendered(TMP_Text t)
    {
        if(t==null || !t.isActiveAndEnabled || !t.gameObject.activeInHierarchy || string.IsNullOrWhiteSpace(t.text) || t.color.a<=.01f || t.canvasRenderer.cull) return false;
        foreach(var group in t.GetComponentsInParent<CanvasGroup>()) if(group.alpha<=.01f)return false;
        bool glyph=false;
        if(t.textInfo!=null)for(int i=0;i<t.textInfo.characterCount;i++)if(t.textInfo.characterInfo[i].isVisible){glyph=true;break;}
        if(!glyph)return false;
        var canvas=t.GetComponentInParent<Canvas>();
        var camera=canvas!=null && canvas.renderMode!=RenderMode.ScreenSpaceOverlay ? canvas.worldCamera : null;
        var corners=new Vector3[4];t.rectTransform.GetWorldCorners(corners);
        var low=RectTransformUtility.WorldToScreenPoint(camera,corners[0]);var high=RectTransformUtility.WorldToScreenPoint(camera,corners[2]);
        return new Rect(Mathf.Min(low.x,high.x),Mathf.Min(low.y,high.y),Mathf.Abs(high.x-low.x),Mathf.Abs(high.y-low.y)).Overlaps(Screen.safeArea);
    }
    static TMP_Text Label(Transform parent,string path) => parent.Find(path)?.GetComponent<TMP_Text>();
    static void Observe()
    {
        try
        {
            if(EditorApplication.isPlaying)everPlaying=true;
            if(everPlaying && !EditorApplication.isPlayingOrWillChangePlaymode)
            {
                EditorApplication.update-=Observe;
                Append(new Row{stage="ClosedObserver",setupSeen=setupSeen,bindingSeen=bindingSeen,purposeSeen=purposeSeen,activeControlSeen=activeControlSeen,resultSeen=resultSeen,returnSeen=returnSeen,tracePath=(string)TracePath.Invoke(null,null)});
                Debug.Log("[S005NormalAria1151] observerClosed=1 normalJourneyEvidence=PendingOfflineChecks playerReady=0");
                return;
            }
            if(EditorApplication.timeSinceStartup-began>1740d)
            {
                Append(new Row{stage="ObserverFailure",reason="ReadOnlyObserverDeadline; native harness/ownership preserved"});EditorApplication.update-=Observe;return;
            }
            if(!EditorApplication.isPlaying || EditorApplication.timeSinceStartup<next)return;
            next=EditorApplication.timeSinceStartup+.25d;
            foreach(var view in UnityEngine.Object.FindObjectsByType<QuickCustomScreenView>(FindObjectsInactive.Exclude,FindObjectsSortMode.None))
            {
                if(!view.IsBaseAssaultSetup || view.SelectedScenarioId!="S005")continue;
                var title=Label(view.transform,"SkirmishSetupComposition/OperationPreview/OperationName");
                var objective=Label(view.transform,"SkirmishSetupComposition/BaseAssaultRules/RulesScroll/Content/Objective");
                var roster=Label(view.transform,"SkirmishSetupComposition/BaseAssaultRules/RulesScroll/Content/Roster");
                var economy=Label(view.transform,"SkirmishSetupComposition/BaseAssaultRules/RulesScroll/Content/Economy");
                var intel=Label(view.transform,"SkirmishSetupComposition/BaseAssaultRules/RulesScroll/Content/Intel");
                var summary=Label(view.transform,"SkirmishSetupComposition/BaseAssaultRules/RulesScroll/Content/Opponent");
                var config=view.ReadConfigFromControls();
                Append(new Row{stage=bindingSeen ? "NativeAdjustSetupDestination" : "NativeSetupBriefing",catalog=view.SelectedScenarioId,seed=config.MapSeed,title=Text(title),objective=Text(objective),roster=Text(roster),economy=Text(economy),intel=Text(intel),summary=Text(summary),renderedTitle=Rendered(title),renderedObjective=Rendered(objective),renderedRoster=Rendered(roster),renderedEconomy=Rendered(economy),renderedIntel=Rendered(intel)});
                if(Rendered(title) && Rendered(objective)) {if(bindingSeen)returnSeen=true;else setupSeen=true;}
            }
            var world=World.DefaultGameObjectInjectionWorld;
            if(world==null || !world.IsCreated)return;
            var em=world.EntityManager;
            using(var query=em.CreateEntityQuery(typeof(SkirmishResolvedSetupRecord),typeof(SkirmishExpandedSessionComponent)))
            {
                if(query.CalculateEntityCount()!=1)return;
                var session=query.GetSingletonEntity();var state=em.GetComponentData<SkirmishExpandedSessionComponent>(session);var setup=em.GetComponentObject<SkirmishResolvedSetupRecord>(session).Setup;
                if(setup.CatalogId!="S005")throw new InvalidOperationException("Observed foreign normal session");
                if(!bindingSeen)
                {
                    Append(new Row{stage="ActualResolvedBinding",catalog=setup.CatalogId,definition=setup.DefinitionId,size=setup.SizeId.ToString(),difficulty=setup.DifficultyId.ToString(),army=setup.ArmyProfileId.ToString(),start=setup.StartPackageId.ToString(),readiness=setup.Readiness.ToString(),readinessValue=(int)setup.Readiness,seed=setup.Seed,deadline=setup.DeadlineSeconds,objective=setup.ObjectiveKind.ToString(),sharedFog=setup.SharedFog,developmentFullVision=setup.DevelopmentFullVision,attempt=state.SessionId.ToString(),tracePath=(string)TracePath.Invoke(null,null)});
                    bindingSeen=true;
                }
                if(em.HasBuffer<SkirmishProductionReservation>(session))foreach(var receipt in em.GetBuffer<SkirmishProductionReservation>(session,true))
                {
                    string key=receipt.FactionId+":"+receipt.ReservationId+":"+receipt.Phase+":"+receipt.DeliveredMembers;
                    if(!receiptStates.Add(key))continue;
                    int cost=0;foreach(var overlay in setup.RoleOverlays)if(overlay.RoleKind==receipt.Role){cost=overlay.MaterialsCost;break;}
                    Append(new Row{stage="NativePaidReservation",attempt=state.SessionId.ToString(),reservation=receipt.ReservationId,faction=receipt.FactionId,role=receipt.Role.ToString(),phase=receipt.Phase.ToString(),producer=receipt.ProducerRuntimeId,materialsPaid=receipt.MaterialsPaid,expectedCost=cost,members=receipt.MemberCount,deliveredMembers=receipt.DeliveredMembers});
                }
            }
            if(!UiShellRuntimeGateway.TryReadSkirmish(out var model))return;
            bool active=UiShellRuntimeGateway.ReadAriaPlay().Active;
            foreach(var briefing in UnityEngine.Object.FindObjectsByType<AriaTutorialBriefingView>(FindObjectsInactive.Exclude,FindObjectsSortMode.None))
            {
                if(!briefing.IsPresentationVisible)continue;
                bool title=Rendered(briefing.TitleText),body=Rendered(briefing.BodyText);
                Append(new Row{stage="NativeHudPurpose",title=Text(briefing.TitleText),body=Text(briefing.BodyText),renderedTitle=title,renderedBody=body,ariaActive=active,finished=model.Finished,modelObjective=model.Objective,expanded=model.Expanded,combinedArmsField=model.CombinedArmsField});
                if(title&&body&&!active)purposeSeen=true;
            }
            foreach(var button in UnityEngine.Object.FindObjectsByType<Button>(FindObjectsInactive.Exclude,FindObjectsSortMode.None))
            {
                if(button.name!="WatchAriaPlay" && button.name!="AriaInstantStop")continue;
                bool presented=Presented(button);var label=button.GetComponentInChildren<TMP_Text>();
                Append(new Row{stage="NativeAriaControl",controlPath=PathOf(button.transform),controlText=Text(label),presentedControl=presented,renderedBody=Rendered(label),ariaActive=active,finished=model.Finished});
                if(active&&presented)activeControlSeen=true;
            }
            foreach(var view in UnityEngine.Object.FindObjectsByType<SkirmishMatchView>(FindObjectsInactive.Exclude,FindObjectsSortMode.None))
            {
                var strip=view.transform.Find("BaseObjectives");bool own=false,enemy=false,clock=false;
                if(strip!=null)foreach(var t in strip.GetComponentsInChildren<TMP_Text>())if(Rendered(t))
                { string text=Text(t).ToUpperInvariant();own|=text.Contains("YOUR BASE");enemy|=text.Contains("ENEMY BASE");clock|=text.Contains("TIME LEFT"); }
                Append(new Row{stage="NativeObjectiveStrip",ariaActive=active,finished=model.Finished,renderedPlayerBase=own,renderedEnemyBase=enemy,renderedClock=clock,playerBase=model.PlayerBase,enemyBase=model.EnemyBase,clock=model.Clock,modelObjective=model.Objective});
                if(model.Finished)
                {
                    var result=view.transform.Find("SkirmishResult/CampaignStyleResult");
                    if(result!=null)foreach(var text in result.GetComponentsInChildren<TMP_Text>())if(Rendered(text))
                    {Append(new Row{stage="NativeRenderedResultText",title=Text(text),renderedTitle=true,finished=true});resultSeen=true;}
                }
            }
        }
        catch(Exception error)
        {
            Append(new Row{stage="ObserverFailure",reason=error.ToString()});
            EditorApplication.update-=Observe;
            Debug.LogError("[S005NormalAria1151] observerFailure=1 gameplayUnmodified=1");
        }
    }
}

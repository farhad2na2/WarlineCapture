                if(!em.Exists(hostileTarget)||!em.HasComponent<UnitHealth>(hostileTarget))throw new InvalidOperationException("Native hostile target was disposed before effect attribution.");
                var auditedCohort=AuditCompletePaidCohort(em);
                var events=em.GetBuffer<CombatDamageObservationElement>(observationQueue,true);
                for(int i=0;i<events.Length;i++)
                {
                    var observation=events[i];
                    if(observation.EventId<=eventBaseline||observation.Frame<attackFrame||observation.TargetEntity!=hostileTarget||(observation.SourceEntity!=Entity.Null&&!members.Contains(observation.SourceEntity))||observation.DamageApplied<=0)continue;
                    var expected=CombatDamageSourceKind.DirectFire;
                    if(observation.SourceKind!=expected)throw new InvalidOperationException("Unexpected damage owner for paid role "+roles[index].RoleKind+": "+observation.SourceKind);
                    int current=em.GetComponentData<UnitHealth>(hostileTarget).Current;
                    if(observation.TargetHealthAfter>=targetHealthBefore||current>=targetHealthBefore||current>observation.TargetHealthAfter)throw new InvalidOperationException("Native damage event does not match actual observed target health loss.");
                    var row=new NativeDamageRow{role=roles[index].RoleKind.ToString(),kind=observation.SourceKind.ToString(),targetRole=em.GetComponentData<SkirmishUnitRoleComponent>(hostileTarget).Role.ToString(),targetDomain=targetCategory.ToString(),targetAllowedMask=(int)roles[index].TargetDomains,eventId=observation.EventId,frame=observation.Frame,sourceIndex=observation.SourceEntity.Index,sourceVersion=observation.SourceEntity.Version,targetIndex=hostileTarget.Index,targetVersion=hostileTarget.Version,damageApplied=observation.DamageApplied,targetHealthAfter=observation.TargetHealthAfter,targetHealthBefore=targetHealthBefore,observedCurrentHealth=current,reservation=receiptId,sourcePosition=(Vector3)observation.SourceWorldPosition,targetPosition=(Vector3)observation.TargetWorldPosition,sourceExactPaidMember=observation.SourceEntity!=Entity.Null,sourceEntityIsNull=observation.SourceEntity==Entity.Null,completeHomogeneousPaidCohortVerified=true,attribution=observation.SourceEntity==Entity.Null?"HomogeneousPaidCohortMixedNativeSources":"ExactPaidMember",cohortMembers=members.Count,eligibleDirectFireShooters=eligibleCohortShooters,cohort=auditedCohort,targetExactNativeHostile=true,damageOrEventsInjected=false};
                    File.AppendAllText(effectPath,JsonUtility.ToJson(row)+"\n");
                    Debug.Log("[S005InfantryNativeDamage] result=Passed role="+roles[index].RoleKind+" event="+observation.EventId+" source="+observation.SourceEntity+" paidReservation="+receiptId+" cohortMembers="+members.Count+" soleEligibleDirectFireCohort=1 attribution="+row.attribution+" individualShooterCertified="+(row.sourceExactPaidMember?1:0)+" target="+hostileTarget+" kind="+observation.SourceKind+" damage="+observation.DamageApplied+" healthBefore="+targetHealthBefore+" healthAfter="+observation.TargetHealthAfter+" observedHealth="+current+" shotsProjectilesDamageEventsInjected=0 attackInput=PublicNativeApi selectionInput=Touch");
                    ObserveDiagnostic(em,"QualifiedNativeEffect");
                    SkirmishWorldMovementService.ClearIntent(em,hostileTarget);
                    em.SetComponentData(hostileTarget,hostileOriginalTransform);em.SetComponentData(hostileTarget,hostileOriginalGrid);
                    foreach(var member in members)Kill(em,member);phase=5;caseDeadline=now+10;return;
                }
                if(em.GetComponentData<UnitHealth>(hostileTarget).Current<=0)throw new InvalidOperationException("Hostile target died without an exact paid-source native damage event; no attribution fallback.");
                return;
            }
            if(phase==5)
            {
                foreach(var member in members)if(em.Exists(member) && em.HasComponent<SkirmishAttemptOwnedComponent>(member) && em.GetComponentData<SkirmishAttemptOwnedComponent>(member).CapacityReleased==0)return;
                var cap=em.GetComponentData<SkirmishCapacityComponent>(session);if(cap.InfantryLive+cap.GroundLive+cap.AirLive+cap.SupplyLive!=0)throw new InvalidOperationException("Capacity not fully released after delivered role death.");
                foreach(var member in members)if(em.Exists(member) && SkirmishProductionService.TryReleaseDeath(em,session,member))throw new InvalidOperationException("Death released capacity twice.");
                Debug.Log("[S005InfantryPaidQueueDeath] result=Passed role="+roles[index].RoleKind+" capacityReleased=1 idempotent=1 stimulus=HealthZero");results.Add(roles[index].RoleKind.ToString());index++;phase=2;return;
            }
        }
        catch(Exception e){if(phase==9){EditorApplication.update-=Tick;Debug.LogException(e);completion.TrySetResult(1);}else Finish(true,e.ToString());}
    }

        try
        {
            touch.Tick(Time.unscaledTime);
            if(touch.PlayerInterrupted||touch.UnexpectedSamples!=0||touch.PhysicalInterventions!=0)throw new InvalidOperationException("Native movement Touch audit failed");

            if(touch.IsBusy||now<(double)Get("playerResumeDue")||world==null||!world.IsCreated||phase is not (10 or 11))return;

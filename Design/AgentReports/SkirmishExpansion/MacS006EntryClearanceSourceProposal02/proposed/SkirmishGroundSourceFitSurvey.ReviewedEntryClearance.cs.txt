                    JToken.DeepEquals(x["structureId"],previous["structureId"])).ToArray();
                Require(matches.Length==1&&JToken.DeepEquals(matches[0]["prefabPath"],previous["prefabPath"])&&
                    JToken.DeepEquals(matches[0]["prefabSha256"],previous["prefabSha256"])&&
                    JToken.DeepEquals(matches[0]["resolvedEffectiveRect"],previous["resolvedEffectiveRect"]),
                    "Entry replay changed actual search facility identity/rectangle");
            }
        }
        static void EntryCompareFacilities(JArray expected,JObject row)
        {
            var actual=row["normalFacilityRelocationProposals"] as JArray;
            Require(actual!=null&&actual.Count==expected.Count&&actual.OfType<JObject>().Count()==actual.Count&&
                actual.OfType<JObject>().Select(x=>(int)x["faction"]+"/"+(string)x["structureId"]).Distinct().Count()==actual.Count,
                "Entry incomplete/duplicate allocator roster");
            foreach(JObject previous in expected)
            {
                var next=actual.OfType<JObject>().Single(x=>JToken.DeepEquals(x["faction"],previous["faction"])&&JToken.DeepEquals(x["structureId"],previous["structureId"]));
                Require((bool?)next["entireConfigurationFound"]==true&&JToken.DeepEquals(next["prefabPath"],previous["prefabPath"])&&
                    JToken.DeepEquals(next["prefabSha256"],previous["prefabSha256"]),"Entry incomplete/foreign native facility");
                var a=(JObject)next["resolvedEffectiveRect"];EntryFields(a,"x","z","width","depth");
                Require(a.Properties().All(p=>p.Value.Type==JTokenType.Integer),"Entry noninteger rectangle");
                var rect=new RectInt((int)a["x"],(int)a["z"],(int)a["width"],(int)a["depth"]);
                Require(rect.width>0&&rect.height>0&&!rect.Overlaps(ReviewedEntryRect),"Entry allocator clearance/no-fit refusal");
                bool target=(int)previous["faction"]==1&&(string)previous["structureId"]=="building.ground_staging";
                var old=(JObject)previous["resolvedEffectiveRect"];
                Require(target?JToken.DeepEquals(a["width"],old["width"])&&JToken.DeepEquals(a["depth"],old["depth"]):JToken.DeepEquals(a,old),"Entry unrelated facility/footprint drift refused");
            }
        }
        static void ProbePilotNormalFacilityRelocation(JObject input,JObject[] requests,SkirmishMapLayoutConfig layout,
            SkirmishRoleCatalogConfig roles,EntityManager em,Entity gridEntity,GridConfig grid,MapSurfaceComponent surface,
            JObject report,bool exactTypedRosterCaller=false,bool discoverFacilityPads=false,List<RectInt> plannedApproachClearance=null,
            bool capacityOnly=false,SkirmishScenarioDefinitionConfig serviceAuthority=null)
        {

            ProbePilotNormalFacilityRelocationCore(input,requests,layout,roles,em,gridEntity,grid,surface,report,
                exactTypedRosterCaller,discoverFacilityPads,plannedApproachClearance,capacityOnly,serviceAuthority);
            guard?.AfterProbe(report,capacityOnly); // no postcheck masks an original allocator exception
        }

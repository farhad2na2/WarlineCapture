    static void ObserveOwnTouch(AriaTouchInputUiSystemHelper driver,string kind,Vector2 point)
    {
        if(!ReferenceEquals(driver,touch))return;
        if(kind=="Ended")AriaCommandEvidence.ObserveRelease(Time.frameCount,point);
        File.AppendAllText(touchPath,JsonUtility.ToJson(new NativeTouchRow {frame=Time.frameCount,time=Time.unscaledTime,kind=kind,phase="Runtime",device=driver.DeviceId,generation=driver.Generation,dispatched=driver.DispatchedGestures,completed=driver.CompletedGestures,samples=driver.AcceptedSamples,unexpected=driver.UnexpectedSamples,interventions=driver.PhysicalInterventions,position=point,commandAuditActive=AriaCommandEvidence.Active,acceptedCommands=AriaCommandEvidence.AcceptedCommands,receiptViolations=AriaCommandEvidence.Violations,matchFinished=false})+"\n");
    }
    static void InputFrame()
    {
        if(!EditorApplication.isPlaying||Time.frameCount==inputFrame||touch==null||phase==9)return;
        var visibleTray=UnityEngine.Object.FindAnyObjectByType<MatchHudSquadTrayView>();
        var canvas=visibleTray?.GetComponentInParent<Canvas>()?.rootCanvas;
        if(canvas==null||Mathf.Abs(canvas.pixelRect.width-Screen.width)>1)return;
        inputFrame=Time.frameCount;
        try
        {
            touch.Tick(Time.unscaledTime);
            if(touch.PlayerInterrupted||touch.UnexpectedSamples!=0||touch.PhysicalInterventions!=0)throw new InvalidOperationException("Native movement Touch audit failed");
            double now=EditorApplication.timeSinceStartup;
            if(touch.IsBusy||now<(double)Get("playerResumeDue")||world==null||!world.IsCreated||phase is not (10 or 11))return;
            var em=world.EntityManager;
            if(!UiShellRuntimeGateway.TryReadExpandedSquadPage(out var page)||!UiShellRuntimeGateway.TryReadMatchHudCommandState(out var command))return;
            var tray=UnityEngine.Object.FindAnyObjectByType<MatchHudSquadTrayView>();
            var controls=UnityEngine.Object.FindAnyObjectByType<MatchOverlayCommandControlsView>();
            if(tray==null||controls==null)return;
            if(phase==10)
            {
                using var query=em.CreateEntityQuery(typeof(SelectedUnitTag));using var selected=query.ToEntityArray(Allocator.Temp);
                bool exact=selected.Length==members.Count;foreach(var actor in selected)exact &= members.Contains(actor);
                if(!exact)
                {
                    if(selected.Length>0||page.SelectedGroups>0){Tap(tray.ClearSelectionButton);return;}
                    if(command.ActiveCommandMode!=TacticalCommandMode.Select){Tap(controls.SelectButton);return;}
                    for(int slot=0;slot<5;slot++)if(SkirmishExpandedPresentedOrders.TryGetPresentedMember(em,session,slot,out var member)&&members.Contains(member)){Tap(tray.VisibleCardButton(slot));return;}
                    if(page.NextPage)Tap(tray.NextPageButton);else if(page.PreviousPage)Tap(tray.PreviousPageButton);
                    return;
                }
                Debug.Log("[S005GroundNativeSelection] result=Passed role="+roles[index].RoleKind+" members="+members.Count+" input=Touch selectionInjected=0");phase=11;
            }
            if(command.ActiveCommandMode!=TacticalCommandMode.Move){Tap(controls.MoveButton);return;}
            bool started=(bool)harness.GetMethod("PlayerResumeWorldPointTap",privateStatic).Invoke(null,new object[]{destination,now});
            if(started){phase=12;Debug.Log("[S005GroundNativeMovement] phase=Order role="+roles[index].RoleKind+" destination="+destination+" input=Touch");}
        }
        catch(Exception e){Finish(true,e.ToString());}
    }
    static string Normalize(string key) {
        foreach(var assembly in AppDomain.CurrentDomain.GetAssemblies()) {
            var type=assembly.GetType("Game.Runtime.BuildingDefinitionPrefabSystemHelper",false);
            if(type!=null)return (string)type.GetMethod("NormalizeSpawnableKey",BindingFlags.Static|BindingFlags.Public|BindingFlags.NonPublic).Invoke(null,new object[]{key});
        }
        throw new MissingMemberException("Shared spawnable key normalizer not loaded.");
    }
    static void Kill(EntityManager em,Entity actor){var h=em.GetComponentData<UnitHealth>(actor);h.Current=0;em.SetComponentData(actor,h);}
    static void Finish(bool abort,string reason)
    {
        failed=abort;failure=reason;
        Canvas.willRenderCanvases-=InputFrame; AriaTouchInputUiSystemHelper.Evidence-=ObserveOwnTouch;
        if(touch!=null){if(touch.IsBusy||touch.DispatchedGestures!=touch.CompletedGestures||touch.UnexpectedSamples!=0||touch.PhysicalInterventions!=0){failed=abort=true;reason+=";IncompleteNativeTouchStream";}Debug.Log("[S005GroundMovementInput] dispatched="+touch.DispatchedGestures+" completed="+touch.CompletedGestures+" acceptedCommands="+AriaCommandEvidence.AcceptedCommands+" violations="+AriaCommandEvidence.Violations);touch.Dispose();touch=null;Set("playerResumeTouch",null);AriaCommandEvidence.End();}
        if(changedEnemy && world!=null && world.IsCreated && enemySystem!=SystemHandle.Null)world.Unmanaged.ResolveSystemStateRef(enemySystem).Enabled=enemyEnabled;
        changedEnemy=false;Hold.SetValue(null,0);phase=9;
        Debug.Log("[S005GroundMovement] result="+(abort?"Failed":"Passed")+" cases="+results.Count+" normalInput=0 fullLifecycle=Pending nativeTouchSelectionMovement=Measured attackService=Pending reason="+reason);
        harness.GetMethod("Finish",BindingFlags.Static|BindingFlags.NonPublic).Invoke(null,new object[]{abort,"isolatedGroundPaidQueues:"+reason});
    }
